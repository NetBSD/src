# NetBSD/i386 Bootloader Implementation Guide

**Platform:** x86 32-bit PC (i386)
**CPU:** Intel 80386, 80486, Pentium and compatible
**Purpose:** Complete guide to implementing a BIOS bootloader for i386

---

## Hardware Specifications

### CPU Support
- **80386DX/SX** (1985): 32-bit, 16-32 MHz
- **80486DX/SX** (1989): Integrated FPU, cache
- **Pentium** (1993): Superscalar, 60-200 MHz
- **Pentium Pro/II/III** (1995-1999): Out-of-order execution
- **Modern x86**: Any 32-bit compatible CPU

### Memory Map (Real Mode)

```
Real Mode (16-bit, Initial Boot):
0x00000000 - 0x000003FF  IVT (Interrupt Vector Table) - 1 KB
0x00000400 - 0x000004FF  BIOS Data Area - 256 bytes
0x00000500 - 0x00007BFF  Free conventional memory
0x00007C00 - 0x00007DFF  MBR/Boot sector load address - 512 bytes
0x00007E00 - 0x0007FFFF  Free conventional memory
0x00080000 - 0x0009FFFF  Extended BIOS Data Area (EBDA)
0x000A0000 - 0x000BFFFF  Video RAM - 128 KB
0x000C0000 - 0x000FFFFF  BIOS ROM - 256 KB
0x00100000+              Extended memory (>1 MB)
```

### Memory Map (Protected Mode)

```
Protected Mode (32-bit, After Switch):
0x00000000 - 0x000FFFFF  Low 1 MB (identity mapped)
0x00100000+              Kernel and data
0xC0000000+              Typical kernel virtual address
```

## Boot Process

### Stage 0: BIOS → MBR

**BIOS Actions:**
1. POST (Power-On Self-Test)
2. Initialize hardware
3. Scan boot devices
4. Load sector 0 (MBR) to 0x7C00
5. Jump to 0x7C00

### Stage 1: MBR (Master Boot Record)

**Location:** Sector 0 of disk
**Load Address:** 0x7C00
**Size:** 512 bytes

**MBR Structure:**
```
Offset  Size  Description
------  ----  -----------
0x000   446   Boot code
0x1BE   16    Partition entry 1
0x1CE   16    Partition entry 2
0x1DE   16    Partition entry 3
0x1EE   16    Partition entry 4
0x1FE   2     Boot signature (0x55 0xAA)
```

**Partition Entry Format:**
```c
struct mbr_partition {
    uint8_t  bootflag;      /* 0x80 = active */
    uint8_t  start_head;
    uint8_t  start_sec;     /* Bits 0-5: sector, 6-7: cylinder high */
    uint8_t  start_cyl;     /* Cylinder low byte */
    uint8_t  type;          /* Partition type (0xA9 = NetBSD) */
    uint8_t  end_head;
    uint8_t  end_sec;
    uint8_t  end_cyl;
    uint32_t lba_start;     /* LBA starting sector */
    uint32_t lba_size;      /* Size in sectors */
} __attribute__((packed));
```

**Minimal MBR Code:**
```asm
/*
 * Simple MBR that loads PBR from active partition
 * Assembled to exactly 512 bytes
 */

    .code16
    .text
    .org 0x7C00

start:
    /* Set up segments */
    xorw    %ax, %ax
    movw    %ax, %ds
    movw    %ax, %es
    movw    %ax, %ss
    movw    $0x7C00, %sp

    /* Relocate MBR to 0x0600 */
    movw    $0x0600, %di
    movw    $0x7C00, %si
    movw    $256, %cx           /* 512 bytes = 256 words */
    cld
    rep movsw

    /* Jump to relocated code */
    ljmp    $0, $relocated

relocated:
    /* Scan partition table for active partition */
    movw    $partition_table, %si
    movb    $4, %cl             /* 4 partition entries */

scan_loop:
    cmpb    $0x80, (%si)        /* Check bootable flag */
    je      found_active
    addw    $16, %si            /* Next entry */
    decb    %cl
    jnz     scan_loop
    
    /* No active partition found */
    movw    $msg_no_active, %si
    call    print_string
    jmp     halt

found_active:
    /* Load PBR from partition */
    movl    8(%si), %eax        /* Get LBA start */
    movw    $0x7C00, %bx        /* Load address */
    
    /* Try LBA read first */
    movw    $lba_packet, %si
    movl    %eax, 8(%si)        /* Set LBA */
    movb    $0x42, %ah          /* Extended read */
    movb    $0x80, %dl          /* Drive 0 */
    int     $0x13
    jnc     pbr_loaded

    /* Fall back to CHS read */
    /* ... CHS conversion code ... */

pbr_loaded:
    /* Verify boot signature */
    cmpw    $0xAA55, 0x7DFE
    jne     invalid_pbr

    /* Jump to PBR */
    ljmp    $0, $0x7C00

invalid_pbr:
    movw    $msg_invalid, %si
    call    print_string
halt:
    hlt
    jmp     halt

print_string:
    lodsb
    testb   %al, %al
    jz      print_done
    movb    $0x0E, %ah
    int     $0x10
    jmp     print_string
print_done:
    ret

/* Data */
lba_packet:
    .byte   0x10                /* Packet size */
    .byte   0                   /* Reserved */
    .word   1                   /* Sectors to read */
    .word   0x7C00              /* Offset */
    .word   0                   /* Segment */
    .quad   0                   /* LBA (filled in) */

msg_no_active:
    .asciz  "No active partition\r\n"
msg_invalid:
    .asciz  "Invalid boot sector\r\n"

/* Pad to partition table */
    .org 0x1BE - 0x600
partition_table:
    .space  64                  /* 4 × 16-byte entries */
    .word   0xAA55              /* Boot signature */

### Stage 2: PBR (Partition Boot Record)

**Location:** First sector of NetBSD partition
**Load Address:** 0x7C00 (overwrites MBR)
**Size:** 512 bytes

**PBR Code:**
```asm
/*
 * NetBSD PBR - loads bootxx
 * File: /sys/arch/i386/stand/bootxx/pbr.S
 */

    .code16
    .text
    .org 0

start:
    jmp     start0
    nop

    /* BIOS Parameter Block (BPB) space */
    .ascii  "NetBSD60"          /* OEM name */
    /* ... BPB fields ... */

start0:
    /* Set up segments */
    xorw    %cx, %cx
    movw    %cx, %ss
    movw    $0x7C00, %sp
    movw    %cx, %es
    movw    %cx, %ds

    /* Reset disk */
    pushw   %dx                 /* Save drive number */
    int     $0x13               /* AH=0: reset disk */
    popw    %dx

    /* Load bootxx (secondary boot) */
    /* bootxx is at sectors 1-15 of partition */
    
load_bootxx:
    movw    $BOOTADDR, %bx      /* Load address: 0x1000 */
    movw    $BOOTXX_SECTORS, %cx /* Number of sectors */
    movl    partition_lba, %eax /* Starting LBA */
    incl    %eax                /* Skip PBR itself */

read_loop:
    pushw   %cx
    call    read_sector_lba
    popw    %cx
    incl    %eax
    addw    $512, %bx
    loop    read_loop

    /* Verify bootxx magic */
    cmpl    $X86_BOOT_MAGIC_1, BOOTADDR+4
    jne     error

    /* Jump to bootxx */
    movw    partition_lba_low, %si
    movw    partition_lba_high, %di
    ljmp    $0, $BOOTADDR

read_sector_lba:
    /* Use INT 13h Extensions */
    movw    $lba_packet, %si
    movl    %eax, 8(%si)
    movb    $0x42, %ah
    int     $0x13
    ret

error:
    movw    $error_msg, %si
    call    print_string
    hlt

lba_packet:
    .word   0x10
    .word   1                   /* 1 sector */
    .word   0                   /* Offset (filled) */
    .word   0                   /* Segment */
    .quad   0                   /* LBA (filled) */

error_msg:
    .asciz  "Boot error\r\n"

    /* Pad to 512 bytes */
    .org 510
    .word   0xAA55

### Stage 3: bootxx (Secondary Boot)

**Location:** `/sys/arch/i386/stand/bootxx/`
**Load Address:** 0x1000
**Size:** ~7.5 KB (15 sectors)

**bootxx Entry:**
```asm
/*
 * bootxx.S - Secondary bootloader
 * Loads /boot from filesystem
 */

    .code16
ENTRY(bootxx)
    jmp     1f
    .balign 4
ENTRY(bootxx_magic)
    .long   X86_BOOT_MAGIC_1

1:  call    gdt_fixup

    /* Switch to protected mode */
    call    real_to_prot
    .code32

    /* Clear BSS */
    movl    $_end, %ecx
    movl    $__bss_start, %edi
    subl    %edi, %ecx
    xorl    %eax, %eax
    rep     stosb

    /* Call C function */
    movzbl  %dl, %edx           /* Drive number */
    pushl   %edi                /* Partition LBA high */
    pushl   %esi                /* Partition LBA low */
    pushl   %edx                /* Drive */
    call    boot1
    addl    $12, %esp

    /* Should not return */
    call    bios_halt

/*
 * Switch from Real Mode to Protected Mode
 */
real_to_prot:
    .code16
    cli

    /* Load GDT */
    lgdt    gdtdesc

    /* Enable protected mode */
    movl    %cr0, %eax
    orl     $CR0_PE, %eax
    movl    %eax, %cr0

    /* Jump to flush pipeline */
    ljmp    $GSEL(GCODE_SEL, SEL_KPL), $1f

    .code32
1:  movw    $GSEL(GDATA_SEL, SEL_KPL), %ax
    movw    %ax, %ds
    movw    %ax, %es
    movw    %ax, %ss
    ret

/* Global Descriptor Table */
    .align  8
gdt:
    .quad   0                   /* Null descriptor */
    .quad   0x00CF9A000000FFFF  /* Code segment */
    .quad   0x00CF92000000FFFF  /* Data segment */

gdtdesc:
    .word   gdtdesc - gdt - 1
    .long   gdt

**boot1.c - Load /boot:**
```c
/*
 * boot1() - Load /boot from filesystem
 */
const char *
boot1(uint32_t biosdev, uint64_t *sector)
{
    struct stat sb;
    int fd;

    printf("\r\nNetBSD/i386 Primary Bootstrap\r\n");

    /* Initialize BIOS disk interface */
    bios_sector = *sector;
    d.biosdev = biosdev;

    /* Open /boot */
    fd = open("boot", 0);
    if (fd < 0)
        return "Cannot open /boot\r\n";

    /* Get size */
    if (fstat(fd, &sb) < 0)
        return "Cannot stat /boot\r\n";

    /* Load into memory */
    if (read(fd, (void *)SECONDARY_LOAD_ADDRESS, sb.st_size) != sb.st_size)
        return "Cannot read /boot\r\n";

    /* Verify magic */
    if (*(uint32_t *)(SECONDARY_LOAD_ADDRESS + 4) != X86_BOOT_MAGIC_2)
        return "Invalid /boot\r\n";

    return NULL;  /* Success */
}
```

### Stage 4: boot (Tertiary Bootloader)

**Location:** `/sys/arch/i386/stand/boot/`
**Load Address:** 0x10000  
**Size:** ~64 KB

**Features:**
- Interactive menu
- Filesystem access (FFS, EXT2FS, CD9660)
- Kernel loading
- boot.cfg parsing

**main() function:**
```c
void
boot2(int biosdev, uint64_t biossector)
{
    int currname;
    char c;

    /* Initialize console */
    initio(boot_params.bp_consdev);

    printf("NetBSD/i386 BIOS Boot, Revision %s\n", version);

    /* Enable A20 gate */
    gateA20();

    /* Parse boot.cfg */
    parsebootconf(BOOTCFG_FILENAME);

    /* Show boot menu or auto-boot */
    if (bootcfg_info.nummenu > 0) {
        doboottypemenu();
    }

    /* Interactive prompt */
    printf("Press return to boot now, any other key for boot menu\n");
    
    c = awaitkey(bootcfg_info.timeout, 1);
    if (c != '\r' && c != '\n' && c != '\0') {
        bootmenu();
    }

    /* Boot default kernel */
    exec_netbsd("netbsd", 0);
}
```

### Building i386 Bootloader

```bash
# Build tools
cd /usr/src
./build.sh -m i386 tools

# Build boot blocks
cd /usr/src/sys/arch/i386/stand
make

# Output files:
# bootxx/bootxx_ffsv1  - FFS boot blocks
# boot/boot            - Secondary bootloader

# Install
dd if=bootxx_ffsv1 of=/dev/rsd0a bs=512 count=1
installboot -v /dev/rsd0a /usr/mdec/bootxx_ffsv1
```
